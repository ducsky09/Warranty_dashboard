# 0A. Xem danh sách DB
wrangler d1 list

# 0B. Đếm tổng dòng trên REMOTE (⚠ phải có --remote)
wrangler d1 execute warranty-db \
  --database-id 00c1d762-875c-492c-8d6f-66da40dd4b6f \
  --command "SELECT COUNT(*) as total FROM warranty;" \
  --remote

# Nếu báo "no such table" → chưa tạo schema → đi BƯỚC 1
# Nếu total = 0             → chưa import data → đi BƯỚC 2
# Nếu total > 0             → data có nhưng Workers lỗi → đi BƯỚC 4

  --mode full \
  --out /tmp/pending.sql

  --src FINISHED \
  --mode full \
